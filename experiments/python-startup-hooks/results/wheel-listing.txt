import os
import sys


def _record():
    path = os.environ.get('STARTLOG_FILE')
    if not path:
        return
    try:
        with open(path, 'a', encoding='utf-8') as fh:
            fh.write('pid=%d exe=%s argv=%s\n'
                     % (os.getpid(), sys.executable, ' '.join(sys.argv)))
    except OSError:
        pass


_record()
